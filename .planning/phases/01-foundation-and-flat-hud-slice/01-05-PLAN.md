- Item { id: string; layer: string; group: string; name: string; verb: string; target_ts: number | null; kind: "place"|"infrastructure"|"transit_vehicle"|"aircraft"|"vessel"|"self"|"environment"; severity: number; detail: Record<string, unknown> | null; geom: unknown | null; fetched_at: number; stale_after_s: number | null; computed: boolean }
